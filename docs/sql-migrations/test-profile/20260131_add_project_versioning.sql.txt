-- Таблица для хранения истории изменений позиций
CREATE TABLE IF NOT EXISTS public.project_item_history (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  project_item_id UUID NOT NULL REFERENCES public.project_items(id) ON DELETE CASCADE,
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  action TEXT NOT NULL CHECK (action IN ('created', 'updated', 'deleted')),
  -- Старые значения (до изменения)
  old_name TEXT,
  old_quantity NUMERIC,
  old_unit TEXT,
  old_material_price NUMERIC,
  old_labor_price NUMERIC,
  old_final_material_price NUMERIC,
  old_final_labor_price NUMERIC,
  -- Новые значения (после изменения)
  new_name TEXT,
  new_quantity NUMERIC,
  new_unit TEXT,
  new_material_price NUMERIC,
  new_labor_price NUMERIC,
  new_final_material_price NUMERIC,
  new_final_labor_price NUMERIC,
  -- Дополнительная информация
  changed_fields TEXT[] DEFAULT '{}', -- Какие поля изменились
  change_reason TEXT, -- Причина изменения (опционально)
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Таблица для версий проектов (снапшоты)
CREATE TABLE IF NOT EXISTS public.project_versions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  project_id UUID NOT NULL REFERENCES public.projects(id) ON DELETE CASCADE,
  version_number INTEGER NOT NULL,
  created_by UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  name TEXT NOT NULL, -- Название версии (например, "v1.0", "После согласования с клиентом")
  description TEXT, -- Описание изменений в этой версии
  is_manual BOOLEAN DEFAULT FALSE, -- Создано вручную или автоматически
  total_items INTEGER DEFAULT 0,
  total_value NUMERIC DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  
  UNIQUE(project_id, version_number)
);

-- Таблица для хранения снапшотов позиций в каждой версии
CREATE TABLE IF NOT EXISTS public.project_version_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  version_id UUID NOT NULL REFERENCES public.project_versions(id) ON DELETE CASCADE,
  project_item_id UUID NOT NULL, -- ID оригинальной позиции
  name TEXT NOT NULL,
  quantity NUMERIC NOT NULL,
  unit TEXT NOT NULL,
  material_price NUMERIC NOT NULL,
  labor_price NUMERIC NOT NULL,
  final_material_price NUMERIC DEFAULT 0,
  final_labor_price NUMERIC DEFAULT 0,
  catalog_item_id UUID,
  is_assembly_child BOOLEAN DEFAULT FALSE,
  parent_assembly_id UUID,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Включаем RLS
ALTER TABLE public.project_item_history ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.project_versions ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.project_version_items ENABLE ROW LEVEL SECURITY;

-- RLS для project_item_history
CREATE POLICY "project_item_history_select" ON public.project_item_history FOR SELECT TO authenticated
  USING (
    EXISTS (
      SELECT 1 FROM public.project_items pi
      JOIN public.projects p ON p.id = pi.project_id
      WHERE pi.id = project_item_history.project_item_id
      AND (
        p.user_id = auth.uid()
        OR p.team_id IN (SELECT team_id FROM public.team_members WHERE user_id = auth.uid() AND status = 'active')
        OR EXISTS (SELECT 1 FROM public.project_members WHERE project_id = p.id AND user_id = auth.uid())
      )
    )
  );

-- RLS для project_versions
CREATE POLICY "project_versions_select" ON public.project_versions FOR SELECT TO authenticated
  USING (
    EXISTS (
      SELECT 1 FROM public.projects p
      WHERE p.id = project_versions.project_id
      AND (
        p.user_id = auth.uid()
        OR p.team_id IN (SELECT team_id FROM public.team_members WHERE user_id = auth.uid() AND status = 'active')
        OR EXISTS (SELECT 1 FROM public.project_members WHERE project_id = p.id AND user_id = auth.uid())
      )
    )
  );

-- RLS для project_version_items
CREATE POLICY "project_version_items_select" ON public.project_version_items FOR SELECT TO authenticated
  USING (
    EXISTS (
      SELECT 1 FROM public.project_versions pv
      JOIN public.projects p ON p.id = pv.project_id
      WHERE pv.id = project_version_items.version_id
      AND (
        p.user_id = auth.uid()
        OR p.team_id IN (SELECT team_id FROM public.team_members WHERE user_id = auth.uid() AND status = 'active')
        OR EXISTS (SELECT 1 FROM public.project_members WHERE project_id = p.id AND user_id = auth.uid())
      )
    )
  );

-- Индексы
CREATE INDEX IF NOT EXISTS idx_project_item_history_item ON public.project_item_history(project_item_id);
CREATE INDEX IF NOT EXISTS idx_project_item_history_user ON public.project_item_history(user_id);
CREATE INDEX IF NOT EXISTS idx_project_item_history_created ON public.project_item_history(created_at DESC);
CREATE INDEX IF NOT EXISTS idx_project_versions_project ON public.project_versions(project_id);
CREATE INDEX IF NOT EXISTS idx_project_versions_created ON public.project_versions(created_at DESC);
CREATE INDEX IF NOT EXISTS idx_project_version_items_version ON public.project_version_items(version_id);

-- Функция для логирования изменений позиций
CREATE OR REPLACE FUNCTION log_project_item_change()
RETURNS TRIGGER AS $$
DECLARE
  changes TEXT[] := '{}';
  old_record RECORD;
  new_record RECORD;
BEGIN
  -- Определяем старые и новые значения
  IF TG_OP = 'DELETE' THEN
    old_record := OLD;
    new_record := NULL;
  ELSIF TG_OP = 'INSERT' THEN
    old_record := NULL;
    new_record := NEW;
  ELSE -- UPDATE
    old_record := OLD;
    new_record := NEW;
    
    -- Определяем какие поля изменились
    IF OLD.name IS DISTINCT FROM NEW.name THEN
      changes := array_append(changes, 'name');
    END IF;
    IF OLD.quantity IS DISTINCT FROM NEW.quantity THEN
      changes := array_append(changes, 'quantity');
    END IF;
    IF OLD.unit IS DISTINCT FROM NEW.unit THEN
      changes := array_append(changes, 'unit');
    END IF;
    IF OLD.material_price IS DISTINCT FROM NEW.material_price THEN
      changes := array_append(changes, 'material_price');
    END IF;
    IF OLD.labor_price IS DISTINCT FROM NEW.labor_price THEN
      changes := array_append(changes, 'labor_price');
    END IF;
    IF OLD.final_material_price IS DISTINCT FROM NEW.final_material_price THEN
      changes := array_append(changes, 'final_material_price');
    END IF;
    IF OLD.final_labor_price IS DISTINCT FROM NEW.final_labor_price THEN
      changes := array_append(changes, 'final_labor_price');
    END IF;
  END IF;

  -- Записываем в историю
  INSERT INTO public.project_item_history (
    project_item_id,
    user_id,
    action,
    old_name,
    old_quantity,
    old_unit,
    old_material_price,
    old_labor_price,
    old_final_material_price,
    old_final_labor_price,
    new_name,
    new_quantity,
    new_unit,
    new_material_price,
    new_labor_price,
    new_final_material_price,
    new_final_labor_price,
    changed_fields
  ) VALUES (
    COALESCE(NEW.id, OLD.id),
    auth.uid(),
    TG_OP,
    old_record.name,
    old_record.quantity,
    old_record.unit,
    old_record.material_price,
    old_record.labor_price,
    old_record.final_material_price,
    old_record.final_labor_price,
    new_record.name,
    new_record.quantity,
    new_record.unit,
    new_record.material_price,
    new_record.labor_price,
    new_record.final_material_price,
    new_record.final_labor_price,
    changes
  );

  -- Возвращаем результат для триггера
  IF TG_OP = 'DELETE' THEN
    RETURN OLD;
  ELSE
    RETURN NEW;
  END IF;
END;
$$ LANGUAGE plpgsql;

-- Триггеры для логирования изменений
DROP TRIGGER IF EXISTS project_items_history_trigger ON public.project_items;
CREATE TRIGGER project_items_history_trigger
  AFTER INSERT OR UPDATE OR DELETE ON public.project_items
  FOR EACH ROW EXECUTE FUNCTION log_project_item_change();

-- Функция для создания версии проекта
CREATE OR REPLACE FUNCTION create_project_version(
  p_project_id UUID,
  p_version_name TEXT DEFAULT NULL,
  p_description TEXT DEFAULT NULL,
  p_is_manual BOOLEAN DEFAULT FALSE
)
RETURNS UUID AS $$
DECLARE
  v_version_id UUID;
  v_version_number INTEGER;
  v_total_items INTEGER;
  v_total_value NUMERIC;
BEGIN
  -- Получаем следующий номер версии
  SELECT COALESCE(MAX(version_number), 0) + 1
  INTO v_version_number
  FROM public.project_versions
  WHERE project_id = p_project_id;

  -- Считаем статистику
  SELECT 
    COUNT(*),
    COALESCE(SUM((COALESCE(final_material_price, material_price, 0) + COALESCE(final_labor_price, labor_price, 0)) * quantity), 0)
  INTO v_total_items, v_total_value
  FROM public.project_items
  WHERE project_id = p_project_id;

  -- Создаем версию
  INSERT INTO public.project_versions (
    project_id,
    version_number,
    created_by,
    name,
    description,
    is_manual,
    total_items,
    total_value
  ) VALUES (
    p_project_id,
    v_version_number,
    auth.uid(),
    COALESCE(p_version_name, 'v' || v_version_number),
    p_description,
    p_is_manual,
    v_total_items,
    v_total_value
  ) RETURNING id INTO v_version_id;

  -- Копируем все позиции в версию
  INSERT INTO public.project_version_items (
    version_id,
    project_item_id,
    name,
    quantity,
    unit,
    material_price,
    labor_price,
    final_material_price,
    final_labor_price,
    catalog_item_id,
    is_assembly_child,
    parent_assembly_id
  )
  SELECT 
    v_version_id,
    id,
    name,
    quantity,
    unit,
    material_price,
    labor_price,
    final_material_price,
    final_labor_price,
    catalog_item_id,
    is_assembly_child,
    parent_assembly_id
  FROM public.project_items
  WHERE project_id = p_project_id;

  RETURN v_version_id;
END;
$$ LANGUAGE plpgsql;

-- Функция для восстановления проекта из версии
CREATE OR REPLACE FUNCTION restore_project_version(
  p_project_id UUID,
  p_version_id UUID
)
RETURNS BOOLEAN AS $$
DECLARE
  v_project_user_id UUID;
BEGIN
  -- Проверяем права пользователя
  SELECT user_id INTO v_project_user_id
  FROM public.projects
  WHERE id = p_project_id;

  IF v_project_user_id IS NULL THEN
    RAISE EXCEPTION 'Project not found';
  END IF;

  -- Проверяем права (владелец или admin/kierownik в команде)
  IF v_project_user_id != auth.uid() AND NOT EXISTS (
    SELECT 1 FROM public.team_members
    WHERE user_id = auth.uid()
    AND team_id = (SELECT team_id FROM public.projects WHERE id = p_project_id)
    AND role IN ('admin', 'kierownik')
    AND status = 'active'
  ) THEN
    RAISE EXCEPTION 'Insufficient permissions';
  END IF;

  -- Удаляем текущие позиции
  DELETE FROM public.project_items WHERE project_id = p_project_id;

  -- Восстанавливаем позиции из версии
  INSERT INTO public.project_items (
    project_id,
    name,
    quantity,
    unit,
    material_price,
    labor_price,
    final_material_price,
    final_labor_price,
    catalog_item_id,
    is_assembly_child,
    parent_assembly_id,
    created_at,
    updated_at
  )
  SELECT 
    p_project_id,
    name,
    quantity,
    unit,
    material_price,
    labor_price,
    final_material_price,
    final_labor_price,
    catalog_item_id,
    is_assembly_child,
    parent_assembly_id,
    NOW(),
    NOW()
  FROM public.project_version_items
  WHERE version_id = p_version_id;

  -- Обновляем проект
  UPDATE public.projects
  SET updated_at = NOW()
  WHERE id = p_project_id;

  -- Создаем новую версию после восстановления
  PERFORM create_project_version(
    p_project_id,
    'Restored from v' || (SELECT version_number FROM public.project_versions WHERE id = p_version_id),
    'Project restored from version ' || (SELECT version_number FROM public.project_versions WHERE id = p_version_id),
    TRUE
  );

  RETURN TRUE;
END;
$$ LANGUAGE plpgsql;

-- Автоматическое создание версии при значительных изменениях
CREATE OR REPLACE FUNCTION auto_create_version_on_major_changes()
RETURNS TRIGGER AS $$
DECLARE
  v_changes_count INTEGER;
  v_last_version TIMESTAMPTZ;
BEGIN
  -- Считаем изменения за последний час
  SELECT COUNT(*)
  INTO v_changes_count
  FROM public.project_item_history
  WHERE project_item_id IN (
    SELECT id FROM public.project_items WHERE project_id = NEW.id
  )
  AND created_at > NOW() - INTERVAL '1 hour'
  AND action = 'updated';

  -- Проверяем когда была последняя версия
  SELECT MAX(created_at)
  INTO v_last_version
  FROM public.project_versions
  WHERE project_id = NEW.id;

  -- Если было много изменений или последняя версия была давно, создаем новую
  IF v_changes_count >= 10 OR v_last_version IS NULL OR v_last_version < NOW() - INTERVAL '1 day' THEN
    PERFORM create_project_version(
      NEW.id,
      NULL,
      'Auto-saved version after ' || v_changes_count || ' changes',
      FALSE
    );
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Триггер для авто-создания версий
DROP TRIGGER IF EXISTS project_auto_version_trigger ON public.projects;
CREATE TRIGGER project_auto_version_trigger
  AFTER UPDATE ON public.projects
  FOR EACH ROW EXECUTE FUNCTION auto_create_version_on_major_changes();
